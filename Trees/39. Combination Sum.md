```python
class Solution:
    def combinationSum(self, nums: List[int], target: int) -> List[List[int]]:
        res = []

        def dfs(i, total, p):
            if total == target:
                res.append(p.copy())
                return
            
            for j in range(i, len(nums)):
                if total + nums[j] > target:
                    return
                
                p.append(nums[j])
                dfs(j, total + nums[j], p)
                p.pop()
        
        nums.sort()
        dfs(0, 0, [])
        return res
        
```
